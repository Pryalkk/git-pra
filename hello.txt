●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git checkout feature

●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ nano info.txt
●	Commit the changes.
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git add .
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git commit -m "info.txt modified"
●	Switch back to main branch.
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git checkout main
●	Merge the feature branch into main .
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git branch
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git merge feature
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git push origin main
●	Delete the feature branch after merge.
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git branch -d feature
●	harshita@harshita-virtual-machine:~/new_git_clone/Git_lab_remote$ git branch
